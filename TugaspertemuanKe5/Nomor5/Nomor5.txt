package Pertemuan5PBO;

public class Nomor5 {
    public static void main(String[] args) {
        int tinggi = 5;

        System.out.println("       Half Triangle");

        for (int a = 1; a <= tinggi; a++) {
            for (int spasi = tinggi; spasi > a; spasi--) {
                System.out.print(" ");
            }

            for (int b = 1; b <= (2 * a - 1); b++) {
                System.out.print("*");
            }

            System.out.println();
        }

        System.out.println("\n       Full Triangle");

        for (int a = 1; a <= tinggi; a++) {
            int bintang = (a <= 3) ? (2 * a - 1) : (2 * (tinggi - a) + 1);

            for (int spasi = 1; spasi <= (tinggi - bintang / 2 - 1); spasi++) {
                System.out.print(" ");
            }

            for (int b = 1; b <= bintang; b++) {
                System.out.print("*");
            }

            System.out.println();
        }
    }
}
